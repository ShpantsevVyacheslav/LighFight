public static class CurrentBufferBuild {
 const string Folder="Assets/CurrentBufferReview";
 const string Low="Assets/LighFight/Art/Drafts/TripoMasterPlayable/MasterHumanoidCandidate.prefab";
 [Serializable]class Manifest {public string sourceSnapshotSHA,sceneSHA,mode,unity,api,dependencySHA256,projectSettingsSHA256,scope="Mac-only; configuration is not proof of GPU execution";public int numericMode,width=800,height=1000,warmup=120,frames=200;public bool development=true,automaticSingleCamera=true,frameTimingStats=true;}

 static string Hash(string path){using(var sha=SHA256.Create())return BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(path))).Replace("-","").ToLowerInvariant();}
 static void CurrentCodeGate(){
  if(!File.Exists("current-input-snapshot.json"))throw new Exception("Snapshot absent");
  // JSON dictionary is audited by Python immediately before this named build.
  string permit="current-source-verified.txt";if(!File.Exists(permit)||File.ReadAllText(permit).Trim()!=Hash("current-input-snapshot.json"))throw new Exception("Run fixture verify.py before each build");
 }
 static void Build(MeshDeformation mode){
  string allowed=Path.GetFullPath("/Users/macbook1/Documents/ChatGPT/LighFight/.tools/skinning-current100k-review").TrimEnd(Path.DirectorySeparatorChar);
  string actual=Path.GetFullPath(Path.Combine(Application.dataPath,"..")).TrimEnd(Path.DirectorySeparatorChar);
  if(!string.Equals(actual,allowed,StringComparison.Ordinal))throw new Exception("Refusing PlayerSettings mutation outside exact OWN shoe-unity project: "+actual);
  if((new DirectoryInfo(actual).Attributes&FileAttributes.ReparsePoint)!=0)throw new Exception("Refusing symlinked project directory");
  CurrentCodeGate();
  var old=PlayerSettings.meshDeformation;var oldApis=PlayerSettings.GetGraphicsAPIs(BuildTarget.StandaloneOSX);bool oldDefault=PlayerSettings.GetUseDefaultGraphicsAPIs(BuildTarget.StandaloneOSX),oldTiming=PlayerSettings.enableFrameTimingStats;
  int oldWidth=PlayerSettings.defaultScreenWidth,oldHeight=PlayerSettings.defaultScreenHeight;string oldProduct=PlayerSettings.productName;var oldFullscreen=PlayerSettings.fullScreenMode;
  try {
   PlayerSettings.meshDeformation=mode;PlayerSettings.SetUseDefaultGraphicsAPIs(BuildTarget.StandaloneOSX,false);PlayerSettings.SetGraphicsAPIs(BuildTarget.StandaloneOSX,new[]{GraphicsDeviceType.Metal});PlayerSettings.enableFrameTimingStats=true;
   PlayerSettings.defaultScreenWidth=800;PlayerSettings.defaultScreenHeight=1000;PlayerSettings.fullScreenMode=FullScreenMode.Windowed;PlayerSettings.productName="CurrentBufferFixture";
   if(PlayerSettings.meshDeformation!=mode)throw new Exception("Setting readback mismatch");
   Directory.CreateDirectory(Folder);Directory.CreateDirectory("CurrentBufferBuildsV2");
   EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
   var qa=new GameObject("Single automatic render benchmark").AddComponent<CurrentBufferFixture>();
   qa.acceptedPrefab=AssetDatabase.LoadAssetAtPath<GameObject>(Low);
   if(!qa.acceptedPrefab)throw new Exception("Existing reviewed original-density assets required; never rebuild or simplify geometry here");
   foreach(var prefab in new[]{qa.acceptedPrefab}){
    var skin=prefab.GetComponentInChildren<SkinnedMeshRenderer>();if(!skin)throw new Exception("Missing renderer");
    if(skin.sharedMesh.vertexCount<10000||skin.sharedMesh.vertexCount>250000)throw new Exception("Ordinary original100k fixture expected");
    foreach(var mat in skin.sharedMaterials)foreach(var name in mat.GetTexturePropertyNames())if(mat.GetTexture(name) is Texture2D tex){var path=AssetDatabase.GetAssetPath(tex);var importer=AssetImporter.GetAtPath(path) as TextureImporter;if(importer!=null&&importer.textureCompression!=TextureImporterCompression.Uncompressed)throw new Exception("Benchmark requires existing uncompressed inputs: "+path);}
   }
   var dependencies=AssetDatabase.GetDependencies(new[]{Low},true).Where(File.Exists).OrderBy(x=>x,StringComparer.Ordinal).Select(p=>p+" "+Hash(p)).ToArray();
   string depFile="CurrentBufferBuildsV2/input-dependencies-"+mode+".txt";File.WriteAllLines(depFile,dependencies);
   var manifest=new Manifest{sourceSnapshotSHA=Hash("current-input-snapshot.json"),mode=mode.ToString(),numericMode=(int)PlayerSettings.meshDeformation,unity=Application.unityVersion,api="Metal",dependencySHA256=Hash(depFile),projectSettingsSHA256=Hash("ProjectSettings/ProjectSettings.asset")};
   qa.inputSnapshotSHA=Hash("current-input-snapshot.json");qa.buildManifest=JsonUtility.ToJson(manifest,true);qa.buildMode=mode.ToString();
   var cam=new GameObject("Only automatic camera").AddComponent<Camera>();cam.enabled=true;cam.allowHDR=true;cam.orthographic=true;cam.orthographicSize=1.32f;cam.backgroundColor=new Color(.1f,.12f,.15f,0);cam.clearFlags=CameraClearFlags.SolidColor;
   cam.transform.position=new Vector3(.2f,1.16f,6);cam.transform.LookAt(new Vector3(0,1.06f,0));cam.gameObject.AddComponent<CurrentSkinningRenderCounter>();qa.benchmarkCamera=cam;
   var fx=cam.gameObject.AddComponent<FittedFilmicCamera>();fx.filmicShader=Shader.Find("Hidden/LighFight/FittedFilmic");if(!fx.filmicShader)throw new Exception("Original fitted filmic shader missing");
   RenderSettings.ambientMode=AmbientMode.Flat;RenderSettings.ambientLight=new Color(.45f,.43f,.4f);
   var light=new GameObject("Identical key").AddComponent<Light>();light.type=LightType.Directional;light.intensity=1.2f;light.shadows=LightShadows.None;light.transform.rotation=Quaternion.Euler(45,-30,0);
   string scene=Folder+"/Benchmark.unity";EditorSceneManager.SaveScene(EditorSceneManager.GetActiveScene(),scene);AssetDatabase.SaveAssets();
   foreach(var component in new MonoBehaviour[]{qa,cam.GetComponent<CurrentSkinningRenderCounter>()}){
    var script=MonoScript.FromMonoBehaviour(component);if(!script||!EditorUtility.IsPersistent(script)||string.IsNullOrEmpty(AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(script))))throw new Exception("Persistent named script required");
   }
   manifest.sceneSHA=Hash(scene);File.WriteAllText("CurrentBufferBuildsV2/raw-scene-"+mode+".sha256",manifest.sceneSHA);
   File.Copy(scene,"CurrentBufferBuildsV2/scene-"+mode+".unity",true);
   string build="CurrentBufferBuildsV2/"+mode+".app";
   File.WriteAllText("CurrentBufferBuildsV2/manifest-"+mode+".json",JsonUtility.ToJson(manifest,true));
   var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{scene},target=BuildTarget.StandaloneOSX,locationPathName=build,options=BuildOptions.Development|BuildOptions.CleanBuildCache});
   var compiled=Directory.GetFiles(build,"*.dll",SearchOption.AllDirectories).OrderBy(p=>p,StringComparer.Ordinal).Select(p=>p.Substring(build.Length)+" "+Hash(p));File.WriteAllLines("CurrentBufferBuildsV2/compiled-assemblies-"+mode+".sha256",compiled);
   File.WriteAllText("CurrentBufferBuildsV2/source-snapshot-"+mode+".json",File.ReadAllText("current-input-snapshot.json"));
   if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Benchmark build failed "+mode);
   File.WriteAllText("CurrentBufferBuildsV2/build-result-"+mode+".txt","Succeeded\nbytes="+report.summary.totalSize+"\nmode="+PlayerSettings.meshDeformation+"\n");
  }finally{
   PlayerSettings.meshDeformation=old;PlayerSettings.SetGraphicsAPIs(BuildTarget.StandaloneOSX,oldApis);PlayerSettings.SetUseDefaultGraphicsAPIs(BuildTarget.StandaloneOSX,oldDefault);PlayerSettings.enableFrameTimingStats=oldTiming;
   PlayerSettings.defaultScreenWidth=oldWidth;PlayerSettings.defaultScreenHeight=oldHeight;PlayerSettings.fullScreenMode=oldFullscreen;PlayerSettings.productName=oldProduct;AssetDatabase.SaveAssets();
  }
 }
}
